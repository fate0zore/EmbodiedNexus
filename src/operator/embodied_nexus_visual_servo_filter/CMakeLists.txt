cmake_minimum_required(VERSION 3.8)
project(embodied_nexus_visual_servo_filter)
find_package(ament_cmake REQUIRED)
find_package(moveit_core REQUIRED)
find_package(pluginlib REQUIRED)
find_package(rclcpp REQUIRED)
find_package(moveit_ros_planning REQUIRED)
add_library(visual_servo_filter SHARED src/passthrough.cpp)
ament_target_dependencies(visual_servo_filter moveit_core pluginlib rclcpp)
pluginlib_export_plugin_description_file(moveit_core plugins.xml)
install(TARGETS visual_servo_filter LIBRARY DESTINATION lib)
add_executable(collision_probe src/collision_probe.cpp)
ament_target_dependencies(collision_probe moveit_core moveit_ros_planning rclcpp)
install(TARGETS collision_probe DESTINATION lib/${PROJECT_NAME})
ament_package()
